# Pinned test dependencies for the PadSpan HA unit suite.
#
# The suite stubs the entire `homeassistant` package tree (tests/conftest.py),
# so Home Assistant itself is NOT required.  Only pytest + the third-party
# libraries the tested modules import directly are needed.
pytest==9.2.0
pytest-asyncio==1.4.1
# Runs the generated Bright tree's copy of the suite on every core
# (scripts/bright_build.run_suite, which falls back to one process without it).
pytest-xdist==3.8.0
voluptuous==0.16.0
# The IRK resolver's AES path (private_ble_resolver._address_matches_irk).
# It imports cryptography inside a try/except ImportError and returns False on
# failure, so without this the IRK and telemetry tests fail as "key does not
# match" rather than as a missing library.
cryptography==50.0.2
# URL building in ws_wled.py (_url — IPv6 hosts, safe paths). Home Assistant
# always ships it (aiohttp depends on it); the stubbed suite needs it pinned.
yarl==1.25.1
# The config-flow translation test reads services.yaml (Home Assistant ships PyYAML).
pyyaml==6.0.3
# The stairs-from-plans command (ws_house3d_stairs.py) reads and shrinks each
# placed plan with Pillow, which Home Assistant always ships; the stubbed suite
# needs it pinned (tests/test_house3d_stairs.py).
pillow==13.0.0
# dwell_analytics.py buckets days with zoneinfo.ZoneInfo. Windows has no system
# time-zone database, so without this even ZoneInfo("UTC") raises and every
# tests/test_dwell_analytics.py test fails there (Linux CI reads /usr/share/zoneinfo).
tzdata==2026.6
